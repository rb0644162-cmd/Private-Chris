local function jitter(base, variance)
    base = base or 0.01
    variance = variance or 0.005
    return base + (math.random() * variance * 2 - variance)
end

local function updatePara(paraObj, text)
    if not paraObj then return end
    pcall(function()
        if type(paraObj.SetDesc) == "function" then
            paraObj:SetDesc(tostring(text))
            return
        end
        if paraObj.Desc and type(paraObj.Desc) == "userdata" and paraObj.Desc.Text ~= nil then
            paraObj.Desc.Text = tostring(text)
            return
        end
        if paraObj.TextLabel and paraObj.TextLabel.Text ~= nil then
            paraObj.TextLabel.Text = tostring(text)
        end
    end)
end

local function formatNum(p1)
    local n = tonumber(p1) or 0
    local list = {
        { "Qi", 1e18 },
        { "Qa", 1e15 },
        { "T", 1e12 },
        { "B", 1e9 },
        { "M", 1e6 },
        { "K", 1e3 },
    }
    for _, value in ipairs(list) do
        if math.abs(n) >= value[2] then
            return string.format("%.2f%s", n / value[2], value[1])
        end
    end
    return tostring(math.floor(n))
end

local function fmtRate(r)
    return "Per hour: " .. formatNum(r * 3600) ..
        "\nPer day: " .. formatNum(r * 86400) ..
        "\nPer week: " .. formatNum(r * 604800) ..
        "\nPer month: " .. formatNum(r * 2592000)
end

local function formatDuration(elapsed)
    local totalSec = math.max(0, math.floor(elapsed or 0))
    local days = math.floor(totalSec / 86400)
    local hours = math.floor((totalSec % 86400) / 3600)
    local minutes = math.floor((totalSec % 3600) / 60)
    local seconds = math.floor(totalSec % 60)
    return string.format("%dd %dh %dm %ds", days, hours, minutes, seconds)
end

local BossState = { AutoFarmBoss = false, CurrentBoss = "None" }
local BossCache = { Arena = nil, LastCheck = 0 }

local function getBossTargets()
    local bosses, seen = {}, {}
    local myChar = localPlayer.Character
    local function add(m)
        if m and m:IsA("Model") and m ~= myChar and not seen[m] and m:IsDescendantOf(workspaceService) then
            if not SelectedBosses[m.Name] then return end
            seen[m] = true; table.insert(bosses, m)
        end
    end
    if (not BossCache.Arena or not BossCache.Arena.Parent) and tick() - (BossCache.LastCheck or 0) >= 2 then
        BossCache.LastCheck = tick()
        local events = workspaceService:FindFirstChild("Events")
        BossCache.Arena = events and events:FindFirstChild("BossArena")
        if not BossCache.Arena then
            for _, v in ipairs(workspaceService:GetDescendants()) do
                if v.Name == "BossArena" and v:IsA("Folder") then BossCache.Arena = v; break end
            end
        end
    end
    if BossCache.Arena then for _, obj in ipairs(BossCache.Arena:GetChildren()) do if SelectedBosses[obj.Name] then add(obj) end end end
    for _, m in ipairs(game:GetService("CollectionService"):GetTagged("BossEventBoss")) do if SelectedBosses[m.Name] then add(m) end end
    return bosses
end

local function ensureBossHitbox(model)
    if not model then return nil end
    local existing = model:FindFirstChild("Hitbox")
    if existing and existing:IsA("BasePart") then return existing end
    local rootPart = model:FindFirstChild("BossDamageHitbox", true) or model:FindFirstChild("HumanoidRootPart") or model:FindFirstChild("Torso") or model.PrimaryPart or model:FindFirstChildWhichIsA("BasePart")
    if not rootPart then return nil end
    local hitbox = Instance.new("Part")
    hitbox.Name = "Hitbox"; hitbox.Size = Vector3.new(12, 12, 12); hitbox.Transparency = 1
    hitbox.CanCollide = false; hitbox.CanQuery = true; hitbox.CanTouch = true; hitbox.Massless = true; hitbox.Anchored = false
    hitbox.CFrame = rootPart.CFrame
    local weld = Instance.new("WeldConstraint"); weld.Part0 = rootPart; weld.Part1 = hitbox; weld.Parent = hitbox
    hitbox.Parent = model
    return hitbox
end

local function readBossHealth(model)
    local stats = model:FindFirstChild("stats")
    local hp = stats and stats:FindFirstChild("Health")
    local mx = stats and stats:FindFirstChild("MaxHealth")
    if hp and hp:IsA("ValueBase") and type(hp.Value) == "number" then return hp.Value, (mx and tonumber(mx.Value)) or hp.Value end
    local hum = model:FindFirstChildOfClass("Humanoid")
    if hum then return hum.Health, hum.MaxHealth end
    local attr = tonumber(model:GetAttribute("Health"))
    if attr then return attr, tonumber(model:GetAttribute("MaxHealth")) or attr end
    return nil, nil
end

local function bossAttackOnce(model, hitbox)
    local character = localPlayer.Character
    local root = character and character:FindFirstChild("HumanoidRootPart")
    local humanoid = character and character:FindFirstChildOfClass("Humanoid")
    if not root or not humanoid or humanoid.Health <= 0 then return end
    local backpack = localPlayer:FindFirstChild("Backpack")
    local punch = character:FindFirstChild("Punch") or (backpack and backpack:FindFirstChild("Punch"))
    if punch and punch.Parent == backpack then humanoid:EquipTool(punch) end
    local head = model:FindFirstChild("Head", true)
    local headPos = head and head.Position or (hitbox.Position + Vector3.new(0, hitbox.Size.Y * 0.35, 0))
    root.CFrame = CFrame.lookAt(headPos + Vector3.new(0, 8, 0), headPos)
    root.AssemblyLinearVelocity = Vector3.zero; root.AssemblyAngularVelocity = Vector3.zero
    local leftHand = character:FindFirstChild("LeftHand"); local rightHand = character:FindFirstChild("RightHand")
    if leftHand and rightHand then
        pcall(function()
            firetouchinterest(hitbox, rightHand, 0); firetouchinterest(hitbox, rightHand, 1)
            firetouchinterest(hitbox, leftHand, 0); firetouchinterest(hitbox, leftHand, 1)
        end)
    end
    local ev = localPlayer:FindFirstChild("muscleEvent") or localPlayer:FindFirstChild("MuscleEvent")
    if ev then pcall(function() ev:FireServer("punch", "leftHand"); ev:FireServer("punch", "rightHand") end) end
    if punch and punch.Parent == character then pcall(function() punch:Activate() end) end
end

local claimedBossChests = setmetatable({}, { __mode = "k" })
local function findBossChest()
    for _, c in ipairs(game:GetService("CollectionService"):GetTagged("BossEventChest")) do
        if c:IsDescendantOf(workspaceService) and not claimedBossChests[c] then
            local prompt = c:FindFirstChild("bossChestPrompt", true)
            local part = (prompt and prompt.Parent) or (c:IsA("BasePart") and c) or (c:IsA("Model") and (c.PrimaryPart or c:FindFirstChildWhichIsA("BasePart", true)))
            if part and part:IsA("BasePart") then return c, prompt, part end
        end
    end
    for _, d in ipairs(workspaceService:GetDescendants()) do
        if d:IsA("ProximityPrompt") and d.Name == "bossChestPrompt" then
            local p = d.Parent
            if p and p:IsA("Attachment") then p = p.Parent end
            if p and p:IsA("BasePart") and not claimedBossChests[p] then return p, d, p end
        end
    end
    return nil
end

local function claimBossChest(chest, prompt, part)
    claimedBossChests[chest] = true
    task.wait(0.3)
    local char = localPlayer.Character
    local root = char and char:FindFirstChild("HumanoidRootPart")
    if root and part.Parent then pcall(function() root.CFrame = CFrame.lookAt(part.Position + Vector3.new(0, 3, 4), part.Position); root.AssemblyLinearVelocity = Vector3.zero end) end
    task.wait(0.2)
    local attempts = 0
    while BossState.AutoFarmBoss and chest.Parent and attempts < 6 do
        attempts = attempts + 1
        local fired = false
        if prompt and prompt.Parent and type(fireproximityprompt) == "function" then fired = pcall(fireproximityprompt, prompt) end
        if not fired then pcall(function() virtualInputManager:SendKeyEvent(true, Enum.KeyCode.E, false, game); task.wait(0.1); virtualInputManager:SendKeyEvent(false, Enum.KeyCode.E, false, game) end) end
        task.wait(0.5)
    end
end

local function startAutoFarmBoss()
    if BossState.AutoFarmBoss then return end
    BossState.AutoFarmBoss = true
    local lastChestCheck = 0
    task.spawn(function()
        while BossState.AutoFarmBoss do
            pcall(function()
                local chest, prompt, part
                if tick() - lastChestCheck >= 1 then lastChestCheck = tick(); chest, prompt, part = findBossChest() end
                if chest and part and chest:GetAttribute("BossChestEmerging") ~= true then
                    BossState.CurrentBoss = "Claiming chest..."
                    claimBossChest(chest, prompt, part); return
                end
                local char = localPlayer.Character
                local hrp = char and char:FindFirstChild("HumanoidRootPart")
                local target, bestDist = nil, math.huge
                for _, m in ipairs(getBossTargets()) do
                    local hp = readBossHealth(m)
                    if hp == nil or hp > 0 then
                        local part2 = m:FindFirstChild("BossDamageHitbox", true) or m.PrimaryPart or m:FindFirstChildWhichIsA("BasePart", true)
                        local d = (hrp and part2) and (hrp.Position - part2.Position).Magnitude or 0
                        if d < bestDist then target, bestDist = m, d end
                    end
                end
                if not target then BossState.CurrentBoss = "Waiting for boss..."; return end
                local hitbox = target:FindFirstChild("BossDamageHitbox", true)
                if not (hitbox and hitbox:IsA("BasePart")) then hitbox = ensureBossHitbox(target) end
                if not hitbox then return end
                BossState.CurrentBoss = REGULAR_BOSS_FULL[target.Name] or target.Name
                bossAttackOnce(target, hitbox)
            end)
            task.wait(0.01)
        end
    end)
end

local function findAlienBoss()
    local boss = workspaceService:FindFirstChild("Alien Boss")
    if boss and boss:IsA("Model") then
        local hpVal = boss:FindFirstChild("alienBossHealth", true)
        local hum = boss:FindFirstChildOfClass("Humanoid")
        if hpVal and hpVal:IsA("ValueBase") and hpVal.Value <= 0 then return nil end
        if hum and hum.Health <= 0 then return nil end
        return boss
    end
    local globalHp = workspaceService:FindFirstChild("alienBossHealth", true)
    if globalHp and globalHp:IsA("ValueBase") and globalHp.Value > 0 then
        local model = globalHp:FindFirstAncestorOfClass("Model")
        if model then return model end
    end
    for _, obj in ipairs(workspaceService:GetDescendants()) do
        if obj:IsA("Model") then
            local nameLower = string.lower(obj.Name)
            if (string.find(nameLower, "alien boss") or string.find(nameLower, "alienboss")) and not string.find(nameLower, "arena") then
                local hpVal = obj:FindFirstChild("alienBossHealth", true)
                local hum = obj:FindFirstChildOfClass("Humanoid")
                if hpVal and hpVal:IsA("ValueBase") and hpVal.Value <= 0 then
                elseif hum and hum.Health <= 0 then
                else
                    return obj
                end
            end
        end
    end
    return nil
end

local function claimAlienChest()
    local char = getCharacter()
    local root = getRoot()
    if not char or not root then return end
    local chest = workspaceService:FindFirstChild("Alien Chest", true)
    if not chest then
        for _, obj in pairs(workspaceService:GetDescendants()) do
            if (obj:IsA("Model") or obj:IsA("BasePart")) and string.find(string.lower(obj.Name), "chest") then
                chest = obj
                break
            end
        end
    end
    if chest then
        local part = chest:IsA("BasePart") and chest or chest:FindFirstChildWhichIsA("BasePart", true)
        if part then
            root.CFrame = CFrame.new(part.Position + Vector3.new(0, 3, 4))
            task.wait(0.2)
            local prompt = chest:FindFirstChildWhichIsA("ProximityPrompt", true)
            local click = chest:FindFirstChildWhichIsA("ClickDetector", true)
            if prompt then
                local origDist = prompt.MaxActivationDistance
                prompt.MaxActivationDistance = 999999
                pcall(fireproximityprompt, prompt)
                prompt.MaxActivationDistance = origDist
            elseif click then
                pcall(fireclickdetector, click)
            else
                pcall(function()
                    firetouchinterest(part, root, 0)
                    firetouchinterest(part, root, 1)
                end)
            end
        end
    end
    task.wait(0.5)
    for _, btn in ipairs(PlayerGui:GetDescendants()) do
        if btn:IsA("TextButton") and btn.Visible then
            local txt = string.lower(btn.Text)
            if string.find(txt, "open") then
                if getconnections then
                    for _, conn in ipairs(getconnections(btn.MouseButton1Click)) do
                        if conn.Fire then
                            pcall(function() conn:Fire() end)
                        end
                    end
                else
                    local pos = btn.AbsolutePosition
                    local size = btn.AbsoluteSize
                    pcall(function()
                        virtualInputManager:SendMouseButtonEvent(pos.X + size.X / 2, pos.Y + size.Y / 2, 0, true, game, 0)
                        virtualInputManager:SendMouseButtonEvent(pos.X + size.X / 2, pos.Y + size.Y / 2, 0, false, game, 0)
                    end)
                end
            end
        end
    end
end

local function startAlienBossFarm()
    state.AutoFarmAlienBoss = true
    if alienBossTask then task.cancel(alienBossTask) end
    alienBossTask = task.spawn(function()
        local bossWasPresent = false
        while state.AutoFarmAlienBoss do
            pcall(function()
                local boss = findAlienBoss()
                if boss then
                    bossWasPresent = true
                    if getgenv().AlienStatusLbl then
                        updatePara(getgenv().AlienStatusLbl, "Attacking: " .. boss.Name)
                    end
                    local hitbox = boss:FindFirstChild("BossDamageHitbox", true)
                    if not (hitbox and hitbox:IsA("BasePart")) then
                        hitbox = ensureBossHitbox(boss)
                    end
                    if hitbox then
                        bossAttackOnce(boss, hitbox)
                    end
                else
                    if bossWasPresent then
                        if getgenv().AlienStatusLbl then
                            updatePara(getgenv().AlienStatusLbl, "Boss dead! Claiming chest...")
                        end
                        task.wait(1)
                        claimAlienChest()
                        task.wait(2)
                        bossWasPresent = false
                    else
                        if getgenv().AlienStatusLbl then
                            updatePara(getgenv().AlienStatusLbl, "Waiting for boss...")
                        end
                    end
                end
            end)
            task.wait(jitter(0.05, 0.01))
        end
    end)
end

end

local function isExcludedNonPet(nameStr, folderNameStr)
    local n = string.lower(tostring(nameStr or ""))
    local fn = string.lower(tostring(folderNameStr or ""))
    if string.find(fn, "aura") or string.find(fn, "trail") or string.find(fn, "egg") or string.find(fn, "equipped") then
        return true
    end
    local blocked = {
        "aura", "trail", "protein", "egg", "shake", "bar", "blade", "sword",
        "weight", "punch", "pushup", "situp", "gem", "coin", "chest", "key", "potion", "serum"
    }
    for _, word in ipairs(blocked) do
        if string.find(n, word, 1, true) then
            return true
        end
    end
    return false
end

local function isPetEquipped(petInstance)
    local equippedPets = localPlayer:FindFirstChild("equippedPets")
    if not equippedPets then return false end
    for _, slot in ipairs(equippedPets:GetChildren()) do
        if slot == petInstance then
            return true
        end
        for _, desc in ipairs(slot:GetDescendants()) do
            if desc:IsA("ObjectValue") and desc.Value == petInstance then
                return true
            end
        end
    end
    return false
end

local function getOwnedPets()
    local owned = {}
    local petsFolder = localPlayer:FindFirstChild("petsFolder")
    if not petsFolder then return owned end

    for _, folder in ipairs(petsFolder:GetChildren()) do
        if folder:IsA("Folder") and not isExcludedNonPet(folder.Name, folder.Name) then
            for _, pet in ipairs(folder:GetChildren()) do
                if not isExcludedNonPet(pet.Name, folder.Name) then
                    table.insert(owned, pet)
                end
            end
        elseif not isExcludedNonPet(folder.Name, "") then
            table.insert(owned, folder)
        end
    end
    return owned
end

local function getOwnedPetNames()
    local names, seen = {}, {}
    for _, pet in ipairs(getOwnedPets()) do
        if not seen[pet.Name] then
            seen[pet.Name] = true
            table.insert(names, pet.Name)
        end
    end
    table.sort(names)
    return names
end

local function findSellRemote()
    local rEvents = RS:FindFirstChild("rEvents")
    if rEvents then
        for _, name in ipairs({"sellPetEvent", "sellPetRemote", "sellPet", "sellPetsEvent", "sellPetFunction"}) do
            local obj = rEvents:FindFirstChild(name)
            if obj and (obj:IsA("RemoteEvent") or obj:IsA("RemoteFunction")) then
                return obj, "rEvents." .. name
            end
        end
        for _, obj in ipairs(rEvents:GetChildren()) do
            if obj:IsA("RemoteEvent") or obj:IsA("RemoteFunction") then
                local n = string.lower(obj.Name)
                if string.find(n, "sell") and string.find(n, "pet") then
                    return obj, "rEvents." .. obj.Name
                end

        end
    end
    local pkg = RS:FindFirstChild("packages")

        for _, obj in ipairs(net:GetChildren()) do
            if obj:IsA("RemoteEvent") or obj:IsA("RemoteFunction") then

local function findMatchingTool(label)
    local char = localPlayer.Character
    local backpack = localPlayer:FindFirstChild("Backpack")

    local keywords = {}
    if label == "Pesa" or label == "Pesa de Mano" then
        keywords = { "weight" }
    elseif label == "Puño" then
        keywords = { "punch" }
    elseif label == "Abdominales" then
        keywords = { "situps", "situp", "ab" }
    elseif label == "Flexiones" then
        keywords = { "pushups", "pushup", "push" }
    end

    local containers = {}
    if char then table.insert(containers, char) end
    if backpack then table.insert(containers, backpack) end

    for _, container in ipairs(containers) do
        for _, item in ipairs(container:GetChildren()) do
            if item:IsA("Tool") then
                local lowerName = string.lower(item.Name)
                for _, kw in ipairs(keywords) do
                    if string.find(lowerName, kw) then
                        return item
                    end
                end
            end
        end
    end
    return nil
end

local function ensureToolEquipped(char, hum, toolObj)
    if not (char and hum and toolObj) then return end
    if toolObj.Parent ~= char then
        hum:EquipTool(toolObj)
    end
end

local function startToolsLoop()
    if autoExecuteToolsTask then
        task.cancel(autoExecuteToolsTask)
        autoExecuteToolsTask = nil
    end

    autoExecuteToolsTask = task.spawn(function()
        while autoExecuteToolsActive or fastToolsActive do
            pcall(function()
                local char = localPlayer.Character
                local hum = char and char:FindFirstChildOfClass("Humanoid")
                local ev = localPlayer:FindFirstChild("muscleEvent") or localPlayer:FindFirstChild("MuscleEvent")
                local isFast = fastToolsActive
                local burstCount = isFast and 6 or 1

                for _, toolLabel in ipairs(selectedFarmTools) do
                    if not (autoExecuteToolsActive or fastToolsActive) then break end
                    local toolObj = findMatchingTool(toolLabel)
                    if toolObj and char and hum and hum.Health > 0 then
                        ensureToolEquipped(char, hum, toolObj)
                        if toolObj:FindFirstChild("attackTime") then
                            toolObj.attackTime.Value = 0
                        end
                        for _ = 1, burstCount do
                            pcall(function() toolObj:Activate() end)
                        end
                    end

                    if toolLabel == "Puño" and ev then
                        for _ = 1, burstCount do
                            pcall(function()
                                ev:FireServer("punch", "leftHand")
                                ev:FireServer("punch", "rightHand")
                            end)
                        end
                    elseif ev then
                        for _ = 1, burstCount do
                            pcall(function()
                                ev:FireServer("rep")
                            end)
                        end
                    end
                end
            end)
            task.wait(fastToolsActive and 0.01 or 0.1)
        end
    end)
end

local function equipSpecificPets(petList, limit)
    local petsFolder = localPlayer:FindFirstChild("petsFolder")
    if not petsFolder then return end
    local count = 0
    for _, petName in ipairs(petList) do
        for _, folder in pairs(petsFolder:GetChildren()) do
            if folder:IsA("Folder") then
                for _, pet in pairs(folder:GetChildren()) do
                    if pet.Name == petName then
                        replicatedStorage.rEvents.equipPetEvent:FireServer("equipPet", pet)
                        count = count + 1
                        if count >= limit then return end
                    end
                end
            end
        end
    end
end

local function formatCommas(num)
    local str = tostring(math.floor(tonumber(num) or 0))
    return str:reverse():gsub("(%d%d%d)", "%1,"):reverse():gsub("^,", "")
end

local function formatShort(num)
    local n = tonumber(num) or 0
    local suffixes = {
        { 1e18, "Qi" },
        { 1e15, "Qa" },
        { 1e12, "T" },
        { 1e9,  "B" },
        { 1e6,  "M" },
        { 1e3,  "K" },
    }
    for _, s in ipairs(suffixes) do
        if math.abs(n) >= s[1] then
            return string.format("%.2f%s", n / s[1], s[2])
        end
    end
    return tostring(math.floor(n))
end

local function formatStatValue(val)
    if val == nil then return "N/A" end
    if type(val) == "number" then
        return "[ " .. formatCommas(val) .. " | " .. formatShort(val) .. " ]"
    end
    return tostring(val)
end

local function countEquippedPet(plr, petName)
    local equippedPets = plr and plr:FindFirstChild("equippedPets")
    if not equippedPets then return 0 end
    local count = 0
    for _, slot in ipairs(equippedPets:GetChildren()) do
        local ref = slot:FindFirstChild("petReference")
        if ref and ref.Value and ref.Value.Name == petName then
            count = count + 1
        end
    end
    return count
end

local function getEnemyDurabilityHealth(plr)
    if not plr then return 0 end
    local durObj = plr:FindFirstChild("Durability") or (plr:FindFirstChild("leaderstats") and plr.leaderstats:FindFirstChild("Durability"))
    local baseDur = (durObj and tonumber(durObj.Value)) or 0
    local ultimates = plr:FindFirstChild("ultimatesFolder")
    local mult = 1
    if ultimates and ultimates:FindFirstChild("Infernal Health") then
        mult = mult + (0.15 * (tonumber(ultimates["Infernal Health"].Value) or 0))
    end
    return baseDur * mult
end

local wildWizardLbl = statsSection:Paragraph({ Title = "Wild Wizard Bonus", Desc = "Wild Wizard equipped: 0 (0 bonus)" })

local function getMyDamage()
    local strObj = localPlayer:FindFirstChild("leaderstats") and localPlayer.leaderstats:FindFirstChild("Strength")
    if not strObj then
        updatePara(wildWizardLbl, "Wild Wizard equipped: 0 (0 bonus)")
        return 0
    end
    local baseDmg = (tonumber(strObj.Value) or 0) * 0.1
    local wizCount = countEquippedPet(localPlayer, "Wild Wizard")
    local bonusMult = wizCount * 0.33
    local bonusVal = baseDmg * bonusMult
    updatePara(wildWizardLbl, "Wild Wizard equipped: " .. tostring(wizCount) .. " (" .. formatStatValue(bonusVal) .. " bonus)")
    return baseDmg * (1 + bonusMult)
end

local function calcBlowsToKill(enemyHp, myDmg)
    if myDmg <= 0 then return "∞" end
    local blows = math.ceil(enemyHp / myDmg)
    if blows > 50 then return "∞" end
    if blows < 1 then return 1 end
    return blows
end

        "\nGood Karma: " .. formatStatValue(goodVal)

local inBrawlSession = false
local previousState = {
    SavedPosition = nil,
    ActiveToolName = nil
}

local function getBrawlTarget()
    if not inBrawlSession then return nil end

    local myChar = localPlayer.Character
    local myHrp = myChar and myChar:FindFirstChild("HumanoidRootPart")

    for _, p in ipairs(players:GetPlayers()) do
        if p ~= localPlayer and p.Character then
            local hum = p.Character:FindFirstChildOfClass("Humanoid")
            local hrp = p.Character:FindFirstChild("HumanoidRootPart")
            if hum and hum.Health > 0 and hrp then
                local inBrawlFlag = p:FindFirstChild("inBrawl") or p:GetAttribute("inBrawl")
                local isEnemyInBrawl = (inBrawlFlag == true or (typeof(inBrawlFlag) == "Instance" and inBrawlFlag.Value == true))
                
                if isEnemyInBrawl or (myHrp and (myHrp.Position - hrp.Position).Magnitude < 130) then
                    return p
                end
            end
        end
    end
    return nil
end

pcall(function()
    local rEvents = replicatedStorage:FindFirstChild("rEvents")
    local brawlEvent = rEvents and rEvents:FindFirstChild("brawlEvent")
    if brawlEvent and brawlEvent:IsA("RemoteEvent") then
        brawlEvent.OnClientEvent:Connect(function(action)
            if action == "brawlStarting" then
                if getgenv().AutoJoinBrawl and not inBrawlSession then
                    inBrawlSession = true

                    local char = localPlayer.Character
                    local hrp = char and char:FindFirstChild("HumanoidRootPart")
                    local humanoid = char and char:FindFirstChildOfClass("Humanoid")

                    if hrp then
                        previousState.SavedPosition = hrp.CFrame
                    end

                    previousState.ActiveToolName = nil
                    if char then
                        for _, item in ipairs(char:GetChildren()) do
                            if item:IsA("Tool") then
                                previousState.ActiveToolName = item.Name
                                item.Parent = localPlayer:FindFirstChild("Backpack")
                                break
                            end
                        end
                    end

                    if humanoid then
                        humanoid:ChangeState(Enum.HumanoidStateType.Jumping)
                    end

                    pcall(function()
                        brawlEvent:FireServer("joinBrawl")
                    end)
                end

            elseif action == "endBrawl" and inBrawlSession then
                task.wait(1.5)
                local char = localPlayer.Character
                local hrp = char and char:FindFirstChild("HumanoidRootPart")
                
                if hrp and previousState.SavedPosition then
                    hrp.CFrame = previousState.SavedPosition
                    hrp.AssemblyLinearVelocity = Vector3.zero
                end

                task.wait(0.3)

                if previousState.ActiveToolName then
                    local backpack = localPlayer:FindFirstChild("Backpack")
                    local tool = backpack and backpack:FindFirstChild(previousState.ActiveToolName)
                    local humanoid = char and char:FindFirstChildOfClass("Humanoid")
                    if tool and humanoid then
                        humanoid:EquipTool(tool)
                    end
                end

                inBrawlSession = false
            end
        end)
    end
end)

local FullOptGuardado = {
    Activo = false,
    GuisOcultas = {},
    LucesOriginales = {},
    SombrasOriginales = {},
    EfectosOriginales = {},
    ConexionNuevaGui = nil
}

local function esGuiPropiaDelScript(gui)
    if not gui or not gui:IsA("ScreenGui") then
        return false
    end
    local nombre = string.lower(gui.Name)
    if string.find(nombre, "windui") or string.find(nombre, "wind") or string.find(nombre, "chris") or string.find(nombre, "priv") then
        return true
    end
    for _, desc in ipairs(gui:GetDescendants()) do
        if desc:IsA("TextLabel") or desc:IsA("TextButton") then
            if desc.Text == "Christian Priv" or desc.Text == "Priv Chris" or desc.Text == "Full Optimization" then
                return true
            end
        end
    end
    return false
end

local function FullOptimization(estado)
    if estado then
        if FullOptGuardado.Activo then return end
        FullOptGuardado.Activo = true

        table.clear(FullOptGuardado.GuisOcultas)
        for _, gui in ipairs(PlayerGui:GetChildren()) do
            if gui:IsA("ScreenGui") and gui.Enabled and not esGuiPropiaDelScript(gui) then
                FullOptGuardado.GuisOcultas[gui] = true
                gui.Enabled = false
            end
        end

        FullOptGuardado.ConexionNuevaGui = PlayerGui.ChildAdded:Connect(function(gui)
            if FullOptGuardado.Activo and gui:IsA("ScreenGui") then
                task.wait()
                if not esGuiPropiaDelScript(gui) and gui.Enabled then
                    FullOptGuardado.GuisOcultas[gui] = true
                    gui.Enabled = false
                end
            end
        end)

        FullOptGuardado.LucesOriginales = {
            Brightness = lighting.Brightness,
            ClockTime = lighting.ClockTime,
            GlobalShadows = lighting.GlobalShadows,
            FogEnd = lighting.FogEnd,
            Ambient = lighting.Ambient,
            OutdoorAmbient = lighting.OutdoorAmbient
        }

        lighting.Brightness = 0
        lighting.ClockTime = 0
        lighting.GlobalShadows = false
        lighting.FogEnd = 100
        lighting.Ambient = Color3.fromRGB(15, 15, 25)
        lighting.OutdoorAmbient = Color3.fromRGB(15, 15, 25)

        local filtro = lighting:FindFirstChild("OptimizacionOscura")
        if not filtro then
            filtro = Instance.new("ColorCorrectionEffect")
            filtro.Name = "OptimizacionOscura"
            filtro.Parent = lighting
        end
        filtro.Brightness = -0.35
        filtro.Contrast = 0.1
        filtro.Saturation = -0.4
        filtro.Enabled = true

        table.clear(FullOptGuardado.SombrasOriginales)
        table.clear(FullOptGuardado.EfectosOriginales)

        for _, obj in ipairs(workspaceService:GetDescendants()) do
            if obj:IsA("BasePart") and obj.CastShadow then
                FullOptGuardado.SombrasOriginales[obj] = true
                obj.CastShadow = false
            elseif (obj:IsA("ParticleEmitter") or obj:IsA("Trail") or obj:IsA("Beam")) and obj.Enabled then
                FullOptGuardado.EfectosOriginales[obj] = true
                obj.Enabled = false
            end
        end
    else
        if not FullOptGuardado.Activo then return end
        FullOptGuardado.Activo = false

        if FullOptGuardado.ConexionNuevaGui then
            FullOptGuardado.ConexionNuevaGui:Disconnect()
            FullOptGuardado.ConexionNuevaGui = nil
        end

        for gui, _ in pairs(FullOptGuardado.GuisOcultas) do
            if gui and gui.Parent then
                gui.Enabled = true
            end
        end
        table.clear(FullOptGuardado.GuisOcultas)

        if next(FullOptGuardado.LucesOriginales) then
            lighting.Brightness = FullOptGuardado.LucesOriginales.Brightness
            lighting.ClockTime = FullOptGuardado.LucesOriginales.ClockTime
            lighting.GlobalShadows = FullOptGuardado.LucesOriginales.GlobalShadows
            lighting.FogEnd = FullOptGuardado.LucesOriginales.FogEnd
            lighting.Ambient = FullOptGuardado.LucesOriginales.Ambient
            lighting.OutdoorAmbient = FullOptGuardado.LucesOriginales.OutdoorAmbient
        end

        local filtro = lighting:FindFirstChild("OptimizacionOscura")
        if filtro then
            filtro.Enabled = false
        end

        for obj, _ in pairs(FullOptGuardado.SombrasOriginales) do
            if obj and obj.Parent then
                obj.CastShadow = true
            end
        end
        table.clear(FullOptGuardado.SombrasOriginales)

        for obj, _ in pairs(FullOptGuardado.EfectosOriginales) do
            if obj and obj.Parent then
                obj.Enabled = true
            end
        end
        table.clear(FullOptGuardado.EfectosOriginales)
    end
end

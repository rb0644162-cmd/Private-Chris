local function formatNumber(_bP)
    if not _bP or _bP == 0 or _bP ~= _bP or _bP == math.huge or _bP == -math.huge then
        return "0"
    end
    local _bQ = {"", "K", "M", "B", "T", "Qa", "Qi", "Sx", "Sp", "Oc", "No", "Dc"}
    local _bR = math.floor(math.log(_bP, 1000))
    if _bR < 1 then
        return tostring(math.floor(_bP))
    end
    local _bS = _bP / math.pow(1000, _bR)
    return string.format("%.2f%s", _bS, _bQ[_bR + 1] or "")
end
local function formatDuration(_bU)
    return string.format("%02dd %02dh %02dm %02ds", math.floor(_bU / 86400), math.floor((_bU % 86400) / 3600), math.floor((_bU % 3600) / 60), math.floor(_bU % 60))
end

local durabilityValue = nil
local function getDurability()
    if not (durabilityValue and durabilityValue.Parent) then
        durabilityValue = nil
        local _cb = LocalPlayer:FindFirstChild("leaderstats")
        local _cc = {LocalPlayer}
        if _cb then
            table.insert(_cc, _cb)
        end
        for _cd, _ce in ipairs({"Durability", "Defense"}) do
            for _cf, _cg in ipairs(_cc) do
                local _ch = _cg:FindFirstChild(_ce)
                if _ch and _ch:IsA("ValueBase") and type(_ch.Value) == "number" then
                    durabilityValue = _ch
                    break
                end
            end
            if durabilityValue then
                break
            end
        end
    end
    if durabilityValue then
        return durabilityValue.Value, durabilityValue
    end
    local _ci = getgenv().RealDurability or 0
    if _ci > 0 then
        return _ci, "hook"
    end
    return 0, nil
end

        if not _db[_dk.instance] then
            table.insert(_di, _dk)

    table.sort(_di, function(_dl, _dm)

    end)
    for _dn, _do in ipairs(_di) do
        if _dc >= MAX_PETS_EQUIPPED then
            break
        end
        if _do.strength > 0 or _do.durability > 0 then

local function sendReps(_dT, _dU, _dV, _dW)
    local _dX = os.clock()
    local _dY = math.clamp(_dX - (_dT.LastAt or _dX), 0, 0.1)
    _dT.LastAt = _dX
    _dT.Tokens = math.min(_dV, (_dT.Tokens or 0) + _dU * _dY)
    local _dZ = 0
    local _ea = LocalPlayer:FindFirstChild("muscleEvent")
    while _dT.Tokens >= 1 and _dW() and _ea do
        _ea:FireServer("rep")
        _dT.Tokens = _dT.Tokens - 1
        _dZ = _dZ + 1
    end
    return _dZ
end
local function runRebirthMode(_ec, _ed, _ee, _ef, _eg)
    local _eh = {TargetSeconds = 3, MaxRate = 50, SendRate = 50, StrengthPerRep = nil}
    local function _ei(_ej)
        return _ec() and _ed() == _ej
    end
    local function _ek(_el)
        local _em = os.clock()
        if _em - _el.SampleAt < 0.25 then
            return
        end
        local _en = getStats()
        local _eo = _en.Str - _el.StartStrength
        if _eo > 0 and _el.Sent > 0 then
            _eh.StrengthPerRep = _eo / _el.Sent
            local _ep = math.max(0, _el.Target - _en.Str)
            local _eq = math.max(0.1, _eh.TargetSeconds - _el.TrainingSeconds)
            _eh.SendRate = math.clamp(_ep / _eh.StrengthPerRep / _eq, 1, _eh.MaxRate)
        elseif _el.TrainingSeconds >= _eh.TargetSeconds then
            _eh.SendRate = _eh.MaxRate
        end
        _el.SampleAt = _em
    end
    local function _er(_es)
        local _et = getStats()
        local _eu = {
            Target = 5000 + _et.Reb * 2550,
            Rebirths = _et.Reb,
            SampleAt = os.clock(),
            StartStrength = _et.Str,
            Sent = 0,
            Tokens = 0,
            LastAt = os.clock(),
            TrainingSeconds = 0
        }
        _eh.SendRate = _eh.StrengthPerRep and math.clamp(math.max(0, _eu.Target - _eu.StartStrength) / _eh.StrengthPerRep / _eh.TargetSeconds, 1, _eh.MaxRate) or _eh.MaxRate
        local _ev = os.clock()
        local function _ew()
            local _ex = getStats()
            return _ei(_es) and _ex.Reb == _eu.Rebirths and _ex.Str < _eu.Target
        end
        while _ew() do
            local _ey = os.clock()
            local _ez = LocalPlayer.Character
            local _eA = _ez and _ez:FindFirstChildOfClass("Humanoid")
            if _eA and _eA.Health > 0 then
                _eu.TrainingSeconds = _eu.TrainingSeconds + math.max(0, _ey - _ev)
                _ek(_eu)
                _eu.Sent = _eu.Sent + sendReps(_eu, _eh.SendRate, 5, _ew)
            else
                _eu.Tokens = 0
                _eu.LastAt = _ey
            end
            _ev = _ey
            RunService.Heartbeat:Wait()
        end
        if not _ei(_es) or getStats().Reb ~= _eu.Rebirths then
            return
        end
        _eg(function()
            return _ei(_es)
        end)
        local _eB = RS:FindFirstChild("rEvents")
        local _eC = _eB and _eB:FindFirstChild("rebirthRemote")
        while _ei(_es) and getStats().Reb == _eu.Rebirths do
            if _eC then
                pcall(function()
                    if _eC:IsA("RemoteFunction") then
                        _eC:InvokeServer("rebirthRequest")

local function formatShort(_hq)
    _hq = tonumber(_hq) or 0
    local _hr = {"", "K", "M", "B", "T", "Qa"}
    local _hs = 0
    while _hq >= 1000 and _hs < #_hr - 1 do
        _hq = _hq / 1000
        _hs = _hs + 1
    end
    if _hs == 0 then
        return tostring(math.floor(_hq))
    end
    return string.format("%.2f%s", _hq, _hr[_hs + 1])
end
local function getBossHealth(_hu)
    local _hv = _hu:FindFirstChild("stats")

    end
    return nil, nil
end

local function findAlienBoss()
    local _iB = workspace:FindFirstChild("Alien Boss")
    if _iB and _iB:IsA("Model") then
        local _iC = _iB:FindFirstChild("alienBossHealth", true)
        local _iD = _iB:FindFirstChildOfClass("Humanoid")
        if _iC and _iC:IsA("ValueBase") and _iC.Value <= 0 then
            return nil
        end
        if _iD and _iD.Health <= 0 then
            return nil
        end
        return _iB
    end
    local _iE = workspace:FindFirstChild("alienBossHealth", true)
    if _iE and _iE:IsA("ValueBase") and _iE.Value > 0 then
        local _iF = _iE:FindFirstAncestorOfClass("Model")
        if _iF then
            return _iF
        end
    end
    for _iG, _iH in ipairs(workspace:GetDescendants()) do
        if _iH:IsA("Model") then
            local _iI = string.lower(_iH.Name)
            if (string.find(_iI, "alien boss") or string.find(_iI, "alienboss")) and not string.find(_iI, "arena") then
                local _iJ = _iH:FindFirstChild("alienBossHealth", true)
                local _iK = _iH:FindFirstChildOfClass("Humanoid")
                if _iJ and _iJ:IsA("ValueBase") and _iJ.Value <= 0 then
                elseif _iK and _iK.Health <= 0 then
                else
                    return _iH
                end
            end
        end
    end
    return nil
end

local function setFFlag(_jZ, _ka)
    local _kb = tostring(_ka)
    local _kc = tonumber(_ka)
    local function _kd(_ke)
        if type(_ke) ~= "function" then
            return false
        end
        if pcall(_ke, _jZ, _kb) then
            return true
        end
        if _kc and pcall(_ke, _jZ, _kc) then
            return true
        end
        return false
    end
    if _kd(setfflag) then
        return true
    end
    if syn and _kd(syn.setfflag) then
        return true
    end
    local _kf, _kg = pcall(getgenv)
    if _kf and _kg and _kd(_kg.setfflag) then
        return true
    end
    local _kh, _ki
    if getfenv then
        _kh, _ki = pcall(getfenv, 0)
        if _kh and _ki and _kd(_ki.setfflag) then
            return true
        end
    end
    if _kd(rawget(_G, "setfflag")) then
        return true
    end
    return false
end
local function applyFlags(_kk)
    local _kl, _km = 0, 0
    for _kn, _ko in pairs(_kk) do
        if setFFlag(_kn, _ko) then
            _kl = _kl + 1
        else
            _km = _km + 1
        end
    end
    return _kl, _km
end
local function setFPSBoost(_kq)
    state.FPSBoost = _kq
    if _kq then
        applyFlags(fpsBoostFlags)
    else
        applyFlags(defaultFlags)
    end
end
